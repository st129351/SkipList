195 Iterator insert(const T& val) {
196 std::vector<Node*> update(mx_lvl+1, head); // vector of the places for insert
197 Node* curr = head;
198
199 // search place for insert
200 for (int i = curr_mx_lvl; i >= 0; i--) {
201 while (curr->next[i] != tail && curr->next[i]->value < val) {
202 curr = curr->next[i]; // last curr on the zero lvl
203 }
204 update[i] = curr;
205 }
206
207 // if need exclude copy:
208
209 // curr = curr->next[0];
210 // // if item already exist in the 0 lvl
211 // if (curr != tail && curr->value == val) {
212 // return Iterator(curr); // initiate ptr in iter
213 // }
214 int new_lvl = WhatLvl(mx_lvl);
215 if (new_lvl > curr_mx_lvl) {
216 for (int i = curr_mx_lvl + 1; i <= new_lvl; i++) {
217 update[i] = head;
218 }
219 curr_mx_lvl = new_lvl;
220 }
221 // create a new node
222 Node* new_node = new Node(val, new_lvl);
223
224 // updating connections
225 for (int i = 0; i <= new_lvl; i++) {
226 new_node->next[i] = update[i]->next[i];
227 new_node->prev[i] = update[i]; // because update[i] - place, after which the insertion is needed
228 update[i]->next[i]->prev[i] = new_node;
229 update[i]->next[i] = new_node;
230 }
231 all_size++;
232 return Iterator(new_node);
233 }

240 bool erase(const T& val) {
241 std::vector<Node*> update(mx_lvl + 1, nullptr);
242 Node* curr = head;
243
244 // search
245 for (int i = curr_mx_lvl; i >= 0; i--) {
246 while (curr->next[i] != tail && curr->next[i]->value < val) {
247 curr = curr->next[i]; // last curr on the zero lvl
248 }
249 update[i] = curr;
250 }
251 curr = curr->next[0];
252
253 // if an item isn't exist
254 if (curr == tail || curr->value != val) {
255 return false;
256 }
257
258 // updating
259 for (int i = 0; i <= curr->lvl; i++) {
260 update[i]->next[i] = curr->next[i];
261 curr->next[i]->prev[i] = update[i];
262 }
263
264 // updating current max level
265 while (curr_mx_lvl > 0 && head->next[curr_mx_lvl] == tail) {
266 curr_mx_lvl--;
267 }
268
269 delete curr; // ptr on deleted item
270 all_size--;
271 return true;
272 }
